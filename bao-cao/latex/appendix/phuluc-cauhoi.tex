\chapter{Bảng câu hỏi phỏng vấn khảo sát}
\label{app:cauhoi}

Mỗi mức khảo sát dùng một nhóm câu hỏi riêng; câu hỏi đóng để lấy số liệu, câu hỏi mở để phát hiện vấn đề \cite[Ch.~4]{tung_qlhttt}.

\section*{Mức thao tác (nhân viên kho)}
\begin{enumerate}
  \item Một lô hàng nhập trung bình mất bao lâu từ lúc xe đến tới lúc hàng lên kệ? (đóng)
  \item Anh/chị chọn chỗ cất hàng dựa vào đâu? (mở)
  \item Khi soạn hàng xuất, khó khăn lớn nhất là gì? (mở)
  \item Một tuần có bao nhiêu lần không tìm thấy hàng trên kệ? (đóng)
\end{enumerate}

\section*{Mức điều phối (thủ kho, kế toán kho)}
\begin{enumerate}
  \item Mỗi ngày có bao nhiêu phiếu nhập, phiếu xuất? Cao điểm vào lúc nào? (đóng)
  \item Khi hàng giao không khớp đơn đặt hàng, quy trình xử lý ra sao? (mở)
  \item Chênh lệch kiểm kê lần gần nhất là bao nhiêu mã hàng? Nguyên nhân thường gặp? (đóng và mở)
  \item Một nghiệp vụ nhập kho phải ghi chép vào những đâu? (mở)
\end{enumerate}

\section*{Mức lãnh đạo (Ban giám đốc)}
\begin{enumerate}
  \item Những thông tin nào về kho anh/chị cần mà hiện chưa có? (mở)
  \item Ngân sách và thời gian dự kiến cho dự án? (đóng)
  \item Kế hoạch mở rộng kho trong 2 -- 3 năm tới? (mở)
\end{enumerate}

\section*{Mức chuyên gia}
\begin{enumerate}
  \item Loại thẻ và đầu đọc RFID nào phù hợp với hàng linh kiện đóng thùng carton và pallet? (mở)
  \item Những yếu tố môi trường nào làm giảm tỷ lệ đọc thẻ? (mở)
\end{enumerate}
